    kLitVertex=lit.vert
    kPhongFragment=phong.frag

    kTerrainVertex=terrain.vert

    kDepthVertex=depth.vert
    kDepthFragment=depth.frag
    kPickVertex=pick.vert
    kPickFragment=pick.frag
    kFlatVertex=flat.vert

    kParticleFragment=particle.frag
    kFullscreenVertex=fullscreen.vert

    kLitVertex
    kPhongFragment
    kPbrFragment
    kWaterVertex
    kWaterFragment
    kDepthVertex
    kPickVertex
    kPickFragment
    kFlatVertex
    kFlatFragment

    kDownsampleFragment
    kUpsampleFragment
    kCompositeFragment
    kFxaaFragment
    kSkyboxVertex
    kSkyboxFragment
    kConvolveVertex
    kConvolveFragment)

rb_compose_gl_shader_header(rabbet_gl_shader_header
    OUTPUT "${CMAKE_BINARY_DIR}/generated/rabbet/render/shaders/GlShaderSources.h"
    NAMESPACE "rb::shaders"
    SHADER_DIR "${rabbet_shader_dir}"
    ENTRIES ${rabbet_gl_shader_entries}
    PRELUDE ${rabbet_gl_shader_prelude}
    DEPENDS
        light_uniforms.glsl
        shadow_functions.glsl
        pbr_functions.glsl)

target_sources(rabbet PRIVATE "${rabbet_gl_shader_header}")

    "${PROJECT_SOURCE_DIR}")
